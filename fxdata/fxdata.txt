// SECTION ORDER IS LOAD-BEARING (bead monhun-ardu-603): fxdata-build.py packs
// sections in source order (v1.15 has no ordering directive; `include` expands
// in place and `raw_t`/`image_t` blobs append as parsed). Every raw_t runtime
// table therefore goes FIRST so its uint24_t offset stays inside the 64 KiB
// window: src/core/fxmem.hpp casts those offsets to 16-bit fake cart pointers.
// Sprite sections are addressed through 24-bit FX::seekData (SpritesU) and may
// live anywhere, so they follow the tables and are free to grow past 64 KiB.
//
// mhEquip is a raw_t read through fxmem.hpp too, so the equipment *catalog*
// belongs on the table side; only the equip/Sprites.txt sheets are
// SpritesU-addressed.

// Core content tables (bead monhun-ardu-42n.1): packed little-endian AVR struct
// layout, generated by tools/gen-fxtables.cpp (make gen). Symbols land in
// fxdata.h as mhWeaponDefs / mhMonsterAttacks / mhMonsterDefs.
raw_t mhWeaponDefs = "tables/weapondefs.bin"
raw_t mhMonsterAttacks = "tables/monsterattacks.bin"
raw_t mhMonsterDefs = "tables/monsterdefs.bin"

// Quarter-wave sine table (bead monhun-ardu-ept): 65 B moved off MCU flash.
// Emitted by tools/gen-fxtables.cpp from the host SIN65 array; read on device
// through mhFxReadI8 (render pass only, between plane blits).
raw_t mhSin65 = "tables/sin65.bin"

// Creature combat blob (bead monhun-ardu-ljj.1): data/skeletons.json +
// data/creatures/*.json compiled by tools/gen-combat.py into the layout from
// docs/creature-framework.md section 8 (header + fixed-size record arrays,
// little-endian, no padding). Offsets land in combat_meta.hpp; packed but not
// yet read at runtime (migration beads 3-5).
raw_t mhCombat = "tables/combat.bin"

// Equipment catalog (bead monhun-ardu-3o9): data/equipment/*.json compiled by
// tools/gen-equipment.py into the layout from docs/equipment-framework.md
// (header + fixed-size item records, little-endian, no padding). Offsets land
// in equip_meta.hpp; read on device through fxmem.hpp during the render pass
// (the eqf.2 part view).
raw_t mhEquip = "tables/equip.bin"

// Screen data (bead monhun-ardu-cgz): data/screens/*.json compiled by
// tools/gen-screens.py into the layout from docs/quests-shops.md (header +
// u16 def offsets + variable ScreenDef/ScreenRow records, little-endian, no
// padding). Offsets land in screen_meta.hpp; read on device through fxmem.hpp
// during the screen scan/render pass (src/screens.hpp).
raw_t mhScreens = "tables/screens.bin"

// Quest defs (bead monhun-ardu-me6): data/quests/*.json compiled by
// tools/gen-quests.py into a fixed 6 B QuestDef record per quest (header +
// records, little-endian, no padding). Offsets land in quest_meta.hpp; read on
// device through fxmem.hpp during the screen scan/render pass (src/quest.hpp)
// for the board rows and the per-hunt kill target.
raw_t mhQuests = "tables/quests.bin"

// Smith upgrade defs (bead monhun-ardu-4ug): data/smith/*.json compiled by
// tools/gen-smith.py into a fixed 7 B UpgradeDef record per tier (header +
// records, little-endian, no padding). Offsets land in smith_meta.hpp; read on
// device through fxmem.hpp when resolving the tier damage/speed multipliers
// (src/smith.hpp) and by the smith screen (src/screens.hpp).
raw_t mhSmith = "tables/smith.bin"

// Room-graph blob (bead monhun-ardu-fie.3): data/map.json compiled by
// tools/gen-zones.py into the header + fixed-size room/door/spawn/prop/heal
// records from docs/map-zones.md (little-endian, no padding). Offsets land in
// zone_meta.hpp; read on device through fxmem.hpp during scene load (fie.4).
raw_t mhZones = "tables/zones.bin"

// Item table (bead monhun-ardu-prg.2): data/items.json compiled by
// tools/gen-items.py into the 8 B header + 5 B item records from the
// items_meta.hpp ABI (little-endian, no padding). Offsets land in
// items_meta.hpp; read on device through fxmem.hpp by src/core/items.hpp
// (itemHeal/itemKind/itemSell). Placed last among the tables so adding it
// only shifts the sprite sections that follow.
raw_t mhItems = "tables/items.bin"

// Armor + skill table (bead monhun-ardu-arm.1): data/skills.json +
// data/armor.json compiled by tools/gen-armor.py into the 8 B header + 18 B
// piece records + 3 B skill records from the armor_meta.hpp ABI
// (little-endian, no padding). Offsets land in armor_meta.hpp; read on device
// through fxmem.hpp by the arm.2/arm.3 engine. Packed after mhItems so adding
// it only shifts the sprite sections that follow.
raw_t mhArmor = "tables/armor.bin"

// Sprite sections: blocks/fonts/equip sheets, addressed with 24-bit
// SpritesU seeks so they may grow past the tables' 64 KiB window. Equipment
// sheets stay after the base sheets; room layer arrays (maps/Sprites.txt, fie.5)
// follow them and are addressed with seekData like any other sprite.
// The opening-menu sheets were deleted with the menu (`hml.2`).
include "blocks/Sprites.txt"
include "fonts/Sprites.txt"
include "equip/Sprites.txt"
include "maps/Sprites.txt"
